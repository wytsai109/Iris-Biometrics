原 C# 計算皮爾森相關係數 (n=253)，JavaScript 版本精準實作相同的代數公式：

原 C# 程式碼 (Correlation)

public float corr(float[] X, float[] Y)
{
    float sum_X = 0, sum_Y = 0, sum_XY = 0;
    float squareSum_X = 0, squareSum_Y = 0;
    for (int i = 0; i < 253; i++)
    {
        sum_X = sum_X + X[i];
        sum_Y = sum_Y + Y[i];
        sum_XY = sum_XY + X[i] * Y[i];
        squareSum_X = squareSum_X + X[i] * X[i];
        squareSum_Y = squareSum_Y + Y[i] * Y[i];
    }
    float a = (float)Math.Sqrt((253 * squareSum_X - sum_X * sum_X) * 
                              (253 * squareSum_Y - sum_Y * sum_Y));
    float corr = (float)(253 * sum_XY - sum_X * sum_Y) / a;
    return corr;
}

新 JavaScript 核心 (iris_engine.js)

pearsonCorrelation(X, Y) {
    const n = Math.min(253, X.length, Y.length);
    let sum_X = 0, sum_Y = 0, sum_XY = 0;
    let squareSum_X = 0, squareSum_Y = 0;

    for (let i = 0; i < n; i++) {
        const xi = X[i], yi = Y[i];
        sum_X += xi;
        sum_Y += yi;
        sum_XY += xi * yi;
        squareSum_X += xi * xi;
        squareSum_Y += yi * yi;
    }

    let a = Math.sqrt((n * squareSum_X - sum_X * sum_X) * 
                      (n * squareSum_Y - sum_Y * sum_Y));
    if (a === 0 || isNaN(a)) a = 1;
    return (n * sum_XY - sum_X * sum_Y) / a;
}

新 JavaScript 核心 (iris_engine.js)

rgbToGray(imageData) {
    const { width, height, data } = imageData;
    const gray = new Uint8ClampedArray(width * height);
    for (let i = 0, j = 0; i < data.length; i += 4, j++) {
        // 標準亮度加權: 0.299R + 0.587G + 0.114B
        gray[j] = Math.round(0.299 * data[i] + 0.587 * data[i + 1] + 0.114 * data[i + 2]);
    }
    return { gray, width, height };
}
